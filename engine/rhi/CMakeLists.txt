# RHI backends are compiled only when their complete development toolchain is
# present. Keep a deterministic headless implementation for tests and servers.
include(CheckIncludeFileCXX)

add_library(nimbus_rhi STATIC
  src/device_factory.cpp
  src/headless_device.cpp
)
target_include_directories(nimbus_rhi PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(nimbus_rhi PUBLIC nimbus_core)
target_compile_features(nimbus_rhi PUBLIC cxx_std_20)

set(NIMBUS_HAS_VULKAN 0)
if(NIMBUS_ENABLE_VULKAN)
  check_include_file_cxx("vulkan/vulkan.h" NIMBUS_VULKAN_HEADER_FOUND)
  if(NIMBUS_VULKAN_HEADER_FOUND)
    if(NIMBUS_ANDROID)
      set(NIMBUS_HAS_VULKAN 1)
      target_link_libraries(nimbus_rhi PUBLIC vulkan android log)
    else()
      find_library(VULKAN_LIBRARY NAMES vulkan)
      if(VULKAN_LIBRARY)
        set(NIMBUS_HAS_VULKAN 1)
        target_link_libraries(nimbus_rhi PUBLIC ${VULKAN_LIBRARY})
      endif()
    endif()
    if(NIMBUS_HAS_VULKAN)
      target_sources(nimbus_rhi PRIVATE src/vulkan_device.cpp)
    else()
      message(STATUS "nimbus: Vulkan backend disabled (Vulkan loader library not found)")
    endif()
  else()
    message(STATUS "nimbus: Vulkan backend disabled (vulkan/vulkan.h not found)")
  endif()
endif()
target_compile_definitions(nimbus_rhi PRIVATE NIMBUS_HAS_VULKAN=${NIMBUS_HAS_VULKAN})

set(NIMBUS_HAS_GLES 0)
if(NIMBUS_ENABLE_GLES)
  check_include_file_cxx("EGL/egl.h" NIMBUS_EGL_HEADER_FOUND)
  check_include_file_cxx("GLES3/gl32.h" NIMBUS_GLES_HEADER_FOUND)
  if(NIMBUS_EGL_HEADER_FOUND AND NIMBUS_GLES_HEADER_FOUND)
    if(NIMBUS_ANDROID)
      set(NIMBUS_HAS_GLES 1)
      target_link_libraries(nimbus_rhi PUBLIC EGL GLESv3 android log)
    else()
      find_library(EGL_LIBRARY NAMES EGL)
      find_library(GLES_LIBRARY NAMES GLESv2 GLESv3)
      if(EGL_LIBRARY AND GLES_LIBRARY)
        set(NIMBUS_HAS_GLES 1)
        target_link_libraries(nimbus_rhi PUBLIC ${EGL_LIBRARY} ${GLES_LIBRARY})
      endif()
    endif()
    if(NIMBUS_HAS_GLES)
      target_sources(nimbus_rhi PRIVATE src/gles_device.cpp)
    else()
      message(STATUS "nimbus: OpenGL ES backend disabled (EGL/GLES libraries not found)")
    endif()
  else()
    message(STATUS "nimbus: OpenGL ES backend disabled (EGL/GLES development headers not found)")
  endif()
endif()
target_compile_definitions(nimbus_rhi PRIVATE NIMBUS_HAS_GLES=${NIMBUS_HAS_GLES})
