# Lua 5.4 script runtime. The implementation uses the stable public C ABI.
add_library(nimbus_script STATIC src/runtime.cpp)
target_include_directories(nimbus_script PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(nimbus_script PUBLIC nimbus_ecs nimbus_core)
target_compile_features(nimbus_script PUBLIC cxx_std_20)
if(NIMBUS_HAS_LUA)
  target_compile_definitions(nimbus_script PRIVATE NIMBUS_HAS_LUA=1)
  target_link_libraries(nimbus_script PUBLIC ${LUA_LIBRARY})
else()
  target_compile_definitions(nimbus_script PRIVATE NIMBUS_HAS_LUA=0 NIMBUS_HAS_DYNAMIC_LUA=1)
  if(UNIX AND NOT ANDROID)
    target_link_libraries(nimbus_script PUBLIC ${CMAKE_DL_LIBS})
  elseif(ANDROID)
    target_link_libraries(nimbus_script PUBLIC dl)
  endif()
endif()
